The desired sum is shown to be $\ln(2)-\frac{3\cdot(\ln(2))^2}{4}-\frac{\pi}{2}+\frac{\pi^{2}}{16}$.

\vspace{5mm}

Let 

\begin{align*}
    I &:= \sum_{n=1}^{\infty} \frac{(-1)^n}{n} \left(\frac{1}{2n-1} - \frac{1}{2n} + \frac{1}{2n+1}-\cdots\right) \\ &= \sum_{n=1}^{\infty} \left(\frac{(-1)^n}{n(2n-1)}-\frac{(-1)^{n}}{n(2n)}+\frac{(-1)^n}{n(2n+1)}-\cdots\right).
\end{align*}

Additionally, let 

\begin{align*}
f(x) &:= \sum_{n=1}^{\infty} \left(\frac{(-1)^n x^{2n-1}}{n(2n-1)}-\frac{(-1)^{n}x^{2n}}{n(2n)}+\frac{(-1)^{n}x^{2n+1}}{n(2n+1)}-\cdots\right) \\ &= \sum_{n=1}^{\infty} (-1)^nx^{2n-2}\left(\frac{ x^{1}}{n(2n-1)}-\frac{x^{2}}{n(2n)}+\frac{x^{3}}{n(2n+1)}-\cdots\right).
\end{align*}

We note that $f(1) = \sum_{n=1}^{\infty} \left(\frac{(-1)^n}{n(2n-1)}-\frac{(-1)^{n}}{n(2n)}+\frac{(-1)^n}{n(2n+1)}-\cdots\right) = I$ and $f(0) = \sum_{n=1}^{\infty} \left(\frac{0}{n(2n-1)}-\frac{0}{n(2n)}+\frac{0}{n(2n+1)}-\cdots\right) = \sum_{n=1}^{\infty}0=0$. 
\vspace{5mm}

Finally, we introduce the parameter $k$ and let 

\begin{align*}
u(x) &:= \sum_{n=1}^{\infty} \left(\frac{(-1)^n x^{2n-2}}{n}-\frac{(-1)^{n}x^{2n-1}}{n}+\frac{(-1)^{n}x^{2n}}{n}-\cdots\right) \\ &= \sum_{n=1}^{\infty} \sum_{k=1}^{\infty}\left(\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right).
\end{align*}

\vspace{5mm}

\textbf{Lemma 1:} $f(x)$ is uniformly convergent on $[0, 1]$. 

\vspace{5mm}

To prove uniform convergence of $f(x)$ over $[0, 1]$, we will show that $f(x)$ satisfies the Weierstrass M-Test.

\vspace{5mm}

To begin, let

\begin{align*}
s_n(x) &:= \left(\frac{ x^{1}}{n(2n-1)}-\frac{x^{2}}{n(2n)}+\frac{x^{3}}{n(2n+1)}-\cdots\right) \\ &= \sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}
\end{align*}

for all $n \geq 1$. From the above definition, it is clear that each $s_n(x)$ can be represented as a power series $\sum_{k=1}^{\infty}a_kx^k$ with $a_k=\frac{(-1)^{k+1}}{n(2n+k-2)}$. Thus, it can be concluded that each $s_n(x)$ is uniformly convergent on its convergence interval. In addition, each $s_n(x)$ is an alternating series. So, the respective convergence interval for any $s_n(x)$ is also   the interval on which it satisfies the alternating series test. Now, 
\begin{align*}
\lim_{k  \to \infty}\left|\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}\right|=
\lim_{k  \to \infty}\left|\frac{x^{k}}{n(2n+k-2)}\right|.
\end{align*}
From this, it is clear that $\displaystyle\lim_{k  \to \infty}\left|\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}\right|=0$ when $-1 \leq x \leq 1$ . Additionally, $\left|\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}\right| > \left|\frac{(-1)^{k+2}\cdot x^{k+1}}{n(2n+k-1)}\right|$ for all $k \ge 1$ $[-1, 0)$ and $(0, 1]$. Moreover, at $x=0$, $\sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}=0$. Thus, each  $s_n(x)$ is uniformly convergent on $[-1,1]$. 

\vspace{5mm}

As proved earlier, when $-1 \leq x \leq 1$, the sequence $\left\{\left|\frac{(-1)^{k+1}\cdot x^{2n+k-2}}{n(2n+k-2)}\right|\right\}$ for $n \in \mathbb{N^*}$ decreases monotonically or is $0$ for all $n \geq 1$. So if we let $S_{2n+1}(x)$ denote the $(2n+1)th$ partial sum of a specific $s_n(x)$ and $S_{2n}(x)$ denote the $(2n)th$, we get that on $[0, 1]$, \newline $\displaystyle \lim_{n \to \infty} S_{2n+1}(x) = \frac{x^{2n-1}}{n(2n-1)}- \left(\frac{x^{2n}}{n(2n)}-\frac{x^{2n+1}}{n(2n+1)}\right)-\cdots \leq \frac{x^{2n-1}}{n(2n-1)}$ and $\displaystyle \lim_{n \to \infty} S_{2n+1}(x) = \left(\frac{x^{2n-1}}{n(2n-1)}-\frac{x^{2n}}{n(2n)}\right) + \left(\frac{x^{2n+1}}{n(2n+1)}-\frac{x^{2n+2}}{n(2n+2)}\right)+\cdots \geq 0$. And, since each $s_n(x)$ converges, the two limits are equal to $s_n(x)$. Thus, $0 \leq s_n(x) \leq \frac{x^{2n-1}}{n(2n-1)}$. 

\vspace{5mm}

As shown earlier, $0 \leq s_n(x) \leq \frac{x}{n(2n-1)}$ on $[0,1]$ for each $n \geq 1$. Additionally, since $x^{2n-2}$ is nonnegative on the interval $[0, 1]$, $0 \leq s_n(x)\cdot x^{2n-2} \le \frac{x^{2n-1}}{n(2n-1)}$ on $[0,1]$. Now, $\displaystyle \lim_{n \to \infty} \frac{x^{2n-1}}{n(2n-1)} = 0$ when $0 \leq x \leq 1$. So, as $n \to \infty$, $s_n(x)\cdot x^{2n-1} \to 0$ on $0 \leq x \le 1$ on the same interval. Furthermore, since $s_n(x) = \left(\frac{ x^{1}}{n(2n-1)}-\frac{x^{2}}{n(2n)}+\frac{x^{3}}{n(2n+1)}-\cdots\right)$ and is uniformly convergent on $[0, 1]$, $f(x) = \sum_{n=1}^{\infty}(-1)^n\cdot s_n(x)\cdot x^{2n-2}$ on the interval $[0,1]$.

\vspace{5mm}

Since $s_n(x)\cdot x^{2n-2} \ge 0$ on $[0, 1]$, $\left|(-1)^{n}s_n(x)\cdot x^{2n-2}\right|= s_n(x)\cdot x^{2n-2}$ on $[0,1]$. Additionally, $s_n(x)\cdot x^{2n-2} \le \frac{x^{2n-1}}{n(2n-1)}\le\frac{1}{n(2n-1)}\le\frac{1}{n^2}$ on $[0, 1]$ for all $n \ge 1$. Thus $\left|(-1)^{n}s_n(x)\cdot x^{2n-2}\right| \le \frac{1}{n^2}$ for all $n \ge 1$ on $[0,1]$. Moreover, since $\sum_{n=1}^{\infty}\frac{1}{n^2}$ converges, $f(x) = \sum_{n=1}^{\infty}(-1)^n\cdot s_n(x)\cdot x^{2n-2}$ uniformly converges on $[0, 1]$ by the Weierstrass M-Test.



\vspace{5mm}

\textbf{Lemma 2:} $u(x)$ is uniformly convergent and $u(x) = f'(x)$ on $[0, 1)$.

\vspace{5mm}

To prove uniform convergence of $u(x)$ over $[0, 1)$, we will show that $u(x)$ satisfies the definition for uniform convergence. The equality of $u(x)$ and $f'(x)$ over $[0,1)$ follows from the uniform convergence of $u(x)$ over $[0,1)$ and the earlier proven uniform convergence of $f(x)$ over $[0, 1]$.

\vspace{5mm}

Term-by-term differentiation of $f(x)$ produces the following sum:

\begin{equation*}
\sum_{n=1}^{\infty}\frac{\mathrm{d}}{\mathrm{d}x}\left((-1)^n\cdot s_n(x)\cdot x^{2n-2}\right)
=\sum_{n=1}^{\infty}(-1)^n(s_n'(x) \cdot x^{2n-2} + s_n(x) \cdot (2n-2)x^{2n-3}).
\end{equation*}

Since $s_n(x)$ is a power series, $s_n'(x)$ is equal to the term-by-term differentiation of $s_n(x)$ and has the same radius of convergence as $s_n(x)$. Thus, 

\begin{align*}
s_n'(x) &= \left(\frac{1}{n(2n-1)}-\frac{2x}{n(2n)}+\frac{3x^2}{n(2n+1)}-\cdots\right) \\ &= \sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot k \cdot x^{k-1}}{n(2n+k-2)}.
\end{align*}

Substituting $s_n(x)$ and $s_n'(x)$ into the term-by-term differentiation of $f(x)$, $s_n'(x) \cdot x^{2n-2} + s_n(x) \cdot (2n-2)x^{2n-3}$ becomes:

\begin{align*}
&\left(\sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot k \cdot x^{k-1}}{n(2n+k-2)}\right) \cdot x^{2n-2} + \left(\sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot x^{k}}{n(2n+k-2)}\right) \cdot (2n-2)x^{2n-3} \\
&= \sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot k \cdot x^{2n+k-3}}{n(2n+k-2)} + \sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot (2n-2) \cdot x^{2n+k-3}}{n(2n+k-2)} \\
&= \sum_{k=1}^{\infty}\frac{(-1)^{k+1}\cdot k \cdot x^{2n+k-3} +(-1)^{k+1}\cdot (2n-2) \cdot x^{2n+k-3}}{n(2n+k-2)} \\
&= \sum_{k=1}^{\infty}\frac{(2n+k-2)(-1)^{k+1} \cdot x^{2n+k-3}}{n(2n+k-2)} \\
&= \sum_{k=1}^{\infty}\frac{ (-1)^{k+1}\cdot x^{2n+k-3}}{n}.
\end{align*}

Thus, 
\begin{align*}
    &\sum_{n=1}^{\infty}\frac{\mathrm{d}}{\mathrm{d}x}\left((-1)^n\cdot s_n(x)\cdot x^{2n-2}\right) \\
    &= \sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \\
    &= u(x).
\end{align*}

Now, $u(x)$ can be arranged such that:

\begin{align*}
    u(x) =\sum_{n=1}^{\infty} \frac{(-1)^nx^{2n-2}}{n}(1-x+x^2-\cdots).
\end{align*}

And on $[0, 1)$,

\begin{align*}
    u(x) =\sum_{n=1}^{\infty} \frac{(-1)^nx^{2n-2}}{n(1+x)}.
\end{align*}

Now, on $[0, 1)$, $x^{2n-2}$ decreases with increasing $n$ or is $0$ for $n \geq 1$. Thus, $\left\{\left|\frac{(-1)^nx^{2n-2}}{n(1+x)}\right|\right\}$ decreases monotonically or is $0$ as $n \to \infty$. By the same reasoning, $\displaystyle \lim_{n \to \infty} \left|\frac{(-1)^nx^{2n-2}}{n(1+x)}\right|  = 0$. Since $\left\{\left|\frac{(-1)^nx^{2n-2}}{n(1+x)}\right|\right\}$ monotonically decreases to 0, $u(x)$ is a convergent alternating series and the alternating series remainder theorem can be applied. That is, if we let $U_N(x)$ denote $\sum_{n=1}^{N} \frac{(-1)^nx^{2n-2}}{n(1+x)}$, $|U_N(x)-u(x)|\le \frac{x^{2N}}{(N+1)(1+x)}$. 

\vspace{5mm}

Now, let $n > N$. Then, on $[0, 1)$, $|U_n(x)-u(x)| \le \frac{x^{2n}}{(n+1)(x+1)} \leq \frac{x^{2N}}{(N+1)(x+1)}$, by the alternating series remainder theorem and the monotone decreasing property of $\left\{\left|\frac{(-1)^nx^{2n-2}}{n(1+x)}\right|\right\}$. Furthermore, on $[0, 1)$, $\frac{x^{2N}}{(N+1)(x+1)} < \frac{1}{2(N+1)} < \frac{1}{N}$. So, on $[0, 1)$, it can be said that for $t > 0$, if $N > \frac{1}{t}$, then $|U_n(x)-u(x)| < t$ for all $n > N$. Thus, $u(x)$ is uniformly convergent on $[0, 1)$ by definition of uniform convergence. Moreover, since $f(x)$ is uniformly convergent on $[0, 1]$, $u(x) = f'(x)$ on $[0, 1)$. From now on, $u(x)$ will be denoted as $f'(x)$.

\vspace{5mm}

\textbf{Solution:}

% Prove that integrating each term provides I, Prove that the swapping of summing and integrating is valid by DCT

\vspace{5mm}

To begin, we note
\begin{align*}
    &\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\int_{0}^{1}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \mathrm{d}x \\
    &= \sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}}{n(2n+k-2)}\\
    &= \sum_{n=1}^{\infty} \left(\frac{(-1)^n}{n(2n-1)}-\frac{(-1)^{n}}{n(2n)}+\frac{(-1)^n}{n(2n+1)}-\cdots\right)\\
    &= I.
\end{align*}

Now, we let $S_K(x):=\sum_{k=1}^{K}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}$ with $0 \leq x < 1$. We observe that each $S_K(x)$ is measurable for all $n\geq1$, since a finite sum of measurable functions is also measurable. Additionally, we observe that the terms of $\left\{\left|\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right|\right\}$ decrease monotonically to 0 as $k \to \infty$ for all $n \geq 1$ on the interval $(0, 1)$. We add that at $x=0$,  the terms of $\left\{\left|\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right|\right\}$ are all 0. From this we can conclude that for odd terms $|S_{2K+1}(x)|$, $1 \geq |S_{2K+1}(x)|$ since \scalebox{0.93}{$1 \geq \left|\frac{ (-1)^{n}\cdot x^{2n-2}}{n}\right| \geq\left|\frac{ (-1)^{n}\cdot x^{2n-2}}{n}-\cdots-\left(\frac{ (-1)^{n}\cdot x^{2n+2K-3}}{n} -\frac{ (-1)^{n}\cdot x^{2n+2K-2}}{n}\right)\right|$}. \newline Additionally, we can conclude that for even terms $|S_{2K}(x)|$, $1\geq|S_{2K}(x)|$ since  \scalebox{0.93}{$1 \geq \left|\frac{ (-1)^{n}\cdot x^{2n-2}}{n}\right| \geq\left|\frac{ (-1)^{n}\cdot x^{2n-2}}{n}-\left(\frac{ (-1)^{n}\cdot x^{2n-1}}{n}-\frac{ (-1)^{n}\cdot x^{2n}}{n}\right)\cdots-\left(\frac{ (-1)^{n}\cdot x^{2n+2K-3}}{n}\right)\right|$}. \newline Thus, $|S_K(x)|\leq1$ for all x on the interval $[0,1)$ and for all $K \geq 1$.  Now, since it has been proven that $|S_K(x)|\leq1$ for almost all x on $[0,1]$ \footnote{In fact, $|S_K(x)| \leq 1$ for all x on $[0,1]$} and for all $K \geq 1$, we can apply the the Dominating Convergence Theorem (since $g(x)=1$ is Lebesgue integrable on $[0,1]$) to conclude that $\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\int_{0}^{1}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \mathrm{d}x=$ \newline $\sum_{n=1}^{\infty}\int_{0}^{1}\left(\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right) \mathrm{d}x$.

\vspace{20mm}

Similarly, we let $f'_N(x):=\sum_{n=1}^{N}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}$. Evaluating the inner series under the assumption $|x|<1$ yields:

\begin{align*}
    f'_N(x) &= \sum_{n=1}^{N}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \\
    &= \sum_{n=1}^{N}\left(\frac{(-1)^{n}x^{2n-2}}{n}\sum_{k=1}^{\infty}(-1)^{k+1}\cdot x^{k-1}\right)\\
    &=\sum_{n=1}^{N} \frac{(-1)^nx^{2n-2}}{n(1+x)}.
\end{align*}

We can then observe that each $f'_N(x)$ is measurable, since a finite sum of measurable functions is also measurable. Additionally, since each $f'_N(x)$ is an alternating series and the first term of each $f'_N(x)=-\frac{1}{1+x}$. We can apply similar logic as used in the previous case to conclude $|f'_N(x)|\leq1$ for all $x$ on $[0,1)$, since $\frac{1}{1+x}\leq1$ when $|x|<1$. Now, since $|f'_N(x)|\leq1$ almost everywhere on $[0,1]$ and $f'_N(x)$ converges pointwise \footnote{Uniform convergence implies pointwise convergence} to $f(x)$ almost everywhere on $[0,1]$, we can apply the the Dominating Convergence Theorem again to conclude that  \scalebox{0.93}{$\sum_{n=1}^{\infty}\int_{0}^{1}\left(\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right) \mathrm{d}x=\int_{0}^{1}\left(\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n}\right) \mathrm{d}x$}.

Thus applying the Dominating Convergence Theorem twice gives \newline \scalebox{0.96}{$\int_{0}^{1}\left(\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \right)\mathrm{d}x = \sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\int_{0}^{1}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \mathrm{d}x = I$}.



\vspace{5mm}
Now, we evaluate $f'(x)$.

\begin{align*}
\textbf{Case 1 ($0 < x <1$)} \\
    f'(x) &= \sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \\
    &= \sum_{n=1}^{\infty}\left(\frac{(-1)^{n}x^{2n-2}}{n}\sum_{k=1}^{\infty}(-1)^{k+1}\cdot x^{k-1}\right)\\
    &=\sum_{n=1}^{\infty} \frac{(-1)^nx^{2n-2}}{n(1+x)} \\
    &= \frac{1}{x^2(x+1)}\sum_{n=1}^{\infty}\frac{(-1)^nx^{2n}}{n} \\
    &= -\frac{1}{x^2(x+1)}\sum_{n=1}^{\infty}\frac{(-1)^{n+1}x^{2n}}{n} \\
    &=  -\frac{\ln(x^2+1)}{x^2(x+1)} \\
\textbf{Case 2 ($x=0$)} \\
    f'(x) &= \sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{ (-1)^{n+k+1}\cdot x^{2n+k-3}}{n} \\
    &= \sum_{n=1}^{\infty}\left(\frac{(-1)^{n}x^{2n-2}}{n}\sum_{k=1}^{\infty}(-1)^{k+1}\cdot x^{k-1}\right)\\
    &=\sum_{n=1}^{\infty} \frac{(-1)^nx^{2n-2}}{n(1+x)} \\
    &= \frac{(-1)^1\cdot0^0}{1}+\sum_{n=2}^{\infty} \frac{(-1)^n0^{2n-2}}{n(1+x)} \\
    &= -1 + 0 \\
    &= -1
\end{align*}

Thus, 
\begin{align*}
f'(x)=\begin{cases} 
     -1 & x = 0 \\
     -\frac{\ln(x^2+1)}{x^2(x+1)} & 0 < x < 1.\\
\end{cases}
\end{align*}

We note that $f'(x)$ is continuous on $[0,1)$.\footnote{Applying L'Hospital's Rule: $\displaystyle\lim_{x \to 0}-\frac{\ln(x^2+1)}{x^2(x+1)}=\lim_{x \to 0}-\frac{2x}{(3x^2+2x)(x^2+1)}\newline=\lim_{x \to 0}-\frac{2}{(3x+2)(x^2+1)}=-1=\displaystyle\lim_{x \to 0^{+}}-\frac{\ln(x^2+1)}{x^2(x+1)}$. Additionally, the uniform convergence of $f'(x)$ on $[0,1)$ implies continuity on $[0,1).$}

\vspace{5mm}

Since removal of a singular point does not affect the integral of a function, it can be said that $\displaystyle I= \int_{0}^{1}f'(x)\mathrm{d}x=\lim_{t \to 0^{+}}\int_{t}^{1}-\frac{\ln(x^2+1)}{x^2(x+1)}\mathrm{d}x$.

\vspace{5mm}

Now, to evaluate the integral, $t$ will be restricted to the interval $(0,1]$ and a parameter $\alpha$, restricted to $[0,\infty)$, will be introduced into the integral such that 

\begin{align*}
   h(\alpha)=-\lim_{t \to 0^{+}}\int_{t}^{1}\frac{\ln(\alpha x^2+1)}{x^2(x+1)}\mathrm{d}x.
\end{align*}

We note that $h(1) = I$ and $h(0)=0$. Additionally, we observe that the integral is uniformly convergent with respect to $\alpha$ \footnote{We know that for every $\epsilon > 0$, there exists $r$ such that $\left|\int_{r}^{1}-\frac{\ln(\alpha x^2+1)}{x^2(x+1)}\mathrm{d}x\right| < \epsilon$ for all $\alpha > 0$ since each of these functions is integrable on the interval $[r, 1]$ with $0 < r < 1$. Thus, $r$ can be as arbitrarily close to $1$ as needed for $\left|\int_{r}^{1}-\frac{\ln(\alpha x^2+1)}{x^2(x+1)}\mathrm{d}x\right| < \epsilon$.}. As such, $\frac{\mathrm{d}}{\mathrm{d}\alpha}\int=\int\frac{\partial}{\partial\alpha}$. So, the limit notation will be temporarily omitted for clarity. 

\vspace{5mm}

Taking the derivative, we get
\begin{align*}
   h'(\alpha)=-\frac{\mathrm{d}}{\mathrm{d}\alpha}\int_{t}^{1}\frac{\ln(\alpha x^2+1)}{x^2(x+1)}\mathrm{d}x.
\end{align*}

Thus,
\begin{align*}
   h'(\alpha)&=-\int_{t}^{1}\frac{\partial}{\partial{\alpha}}\frac{\ln(\alpha x^2+1)}{x^2(x+1)}\mathrm{d}x
   \\ &= -\int_{t}^{1}\frac{1}{{x^2(x+1)}}\cdot\frac{x^2}{\alpha x^2+1}\mathrm{d}x
   \\ &= -\int_{t}^{1}\frac{1}{(\alpha x^2+1)(x+1)}\mathrm{d}x
   \\ &= -\int_{t}^{1}\left(\left(-\frac{\alpha}{1+\alpha}\right)\left(\frac{x-1}{\alpha x^2 +1}\right)+\frac{1}{(\alpha + 1)(x+1)}\right)\mathrm{d}x
   \\ &= \int_{t}^{1}\left(\frac{\alpha}{1+\alpha}\right)\left(\frac{x-1}{\alpha x^2 +1}\right)\mathrm{d}x + -\int_{t}^{1}\frac{1}{(\alpha + 1)(x+1)}\mathrm{d}x
   \\ &= \left(\frac{\alpha}{1+\alpha}\right)\left(\int_{t}^{1}\left(\frac{x}{\alpha x^2 +1}\right)\mathrm{d}x - \int_{t}^{1}\left(\frac{1}{\alpha x^2 +1}\right)\mathrm{d}x\right) \\
   &+-\frac{1}{\alpha + 1}\int_{t}^{1}\frac{1}{x+1}\mathrm{d}x \\
   &= \frac{\alpha}{1+\alpha}\int_{t}^{1}\frac{x}{\alpha x^2 +1}\mathrm{d}x-\frac{\alpha}{1+\alpha}\int_{t}^{1}\frac{1}{\alpha x^2 +1}\mathrm{d}x+\\
   &-\frac{1}{\alpha + 1}\int_{t}^{1}\frac{1}{x+1}\mathrm{d}x
\end{align*}

\vspace{30mm}

Now, we evaluate each integral separately:

To evaluate $(1)$ we rearrange the terms and let $u = \alpha x^2+1$, so $\mathrm{d}u=2\alpha x{\mathrm{d}x}$:
\begin{align*}
   &\frac{\alpha}{1+\alpha}\int_{t}^{1}\frac{x}{\alpha x^2 +1}\mathrm{d}x = \frac{1}{2(1+\alpha)}\int_{t}^{1}\frac{2\alpha x}{\alpha x^2 +1}\mathrm{d}x = \frac{1}{2(1+\alpha)}\int_{t^2\alpha+1}^{\alpha+1}\frac{\mathrm{d}u}{u} \\
   &=\lim_{t \to 0^{+}}\frac{\ln|\alpha x^2 +1|}{2(1+\alpha)}\Bigg|_t^1=\frac{\ln|\alpha+1|}{2(1+\alpha)}=\frac{\ln(\alpha+1)}{2(\alpha+1)} 
\end{align*}

We observe that $\frac{\ln|\alpha+1|}{2(1+\alpha)}=\frac{\ln(\alpha+1)}{2(\alpha+1)}$, since $\alpha\geq0$.

\begin{align*}
   &(2)\\

   &=-\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}
\end{align*}

\end{flalign*}
\begin{align*}
   &-\frac{1}{\alpha + 1}\int_{t}^{1}\frac{1}{x+1}\mathrm{d}x= -\lim_{t \to 0^{+}}\frac{\ln|x+1|}{\alpha+1}\Bigg|_t^1=-\frac{\ln(2)}{\alpha+1}.
\end{align*}

\vspace{30mm}

Recombining the evaluated integrals, we get 

\begin{align*}
   h'(\alpha) &= -\lim_{t \to 0^{+}}\int_{t}^{1}\frac{1}{(\alpha x^2+1)(x+1)}\mathrm{d}x \\
   &= \frac{\ln(\alpha+1)}{2(\alpha+1)}-\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}-\frac{\ln(2)}{\alpha+1}.
\end{align*}

\vspace{5mm}

Now, as noted earlier, $h(1)=I$ and $h(0)=0$. Thus, $h(1)-h(0)=I$ and $\int_0^1h'(\alpha)\mathrm{d}\alpha=I$ since $h'(\alpha)$ is integrable on $0 \leq \alpha \leq 1$.

Thus, 

\begin{align*}
   I &= \int_{0}^{1}\left(\frac{\ln(\alpha+1)}{2(\alpha+1)}-\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}-\frac{\ln(2)}{\alpha+1}\right)\mathrm{d}\alpha \\
   &= \int_{0}^{1}\frac{\ln(\alpha+1)}{2(\alpha+1)}\mathrm{d}\alpha-\int_{0}^{1}\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}\mathrm{d}\alpha-\int_{0}^{1}\frac{\ln(2)}{\alpha+1}\mathrm{d}\alpha.
\end{align*}

\vspace{5mm}

We again evaluate each integral separately:

\begin{flalign*}
    (1)&&
\end{flalign*}

\begin{align*}
    \int_{0}^{1}\frac{\ln(\alpha+1)}{2(\alpha+1)}\mathrm{d}\alpha = \frac{1}{2}\int_{0}^{\ln(2)}u\mathrm{d}u=\frac{u^2}{4}\Bigg|_{0}^{\ln(2)}=\frac{(\ln(2))^2}{4}-0=\frac{(\ln(2))^2}{4}
\end{align*}

\begin{flalign*}
    (2)&&
\end{flalign*}

We let $\alpha=\tan^2(u)$, so that $\mathrm{d}\alpha=2\tan(u)\sec^2(u)$ and

\begin{align*}
    -\int_{0}^{1}\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}\mathrm{d}\alpha&=-2\int_{0}^{\frac{\pi}{4}}\frac{\sqrt{\tan^2(u)}\arctan{\sqrt{\tan^2(u)}}}{1+\tan^2(u)}\tan(u)\sec^2(u)\mathrm{d}u \\ &=-2\int_{0}^{\frac{\pi}{4}}\frac{\tan^2(u)\sec^2(u)\arctan({\tan(u)})}{1+\tan^2(u)}\mathrm{d}u \\ 
    &= -2\int_{0}^{\frac{\pi}{4}}\frac{u\tan^2(u)\sec^2(u)}{sec^2(u)}\mathrm{d}u \\
    &=-2\int_{0}^{\frac{\pi}{4}}u\tan^2(u)\mathrm{d}u \\
    &= -2\int_{0}^{\frac{\pi}{4}}u(\sec^2(u)-1)\mathrm{d}u
\end{align*}

Now, letting $p=u$ and $\mathrm{d}q= \sec^2(u)-1$, we get $\mathrm{d}p= \mathrm{d}u$ and $q=\tan(u)-u$ and integrate by parts such that $\int p\mathrm{d}q = pq - \int q\mathrm{d}p$.

\begin{align*}  
    &-2\int_{0}^{\frac{\pi}{4}}u(\sec^2(u)-1)\mathrm{d}u = -2\left(u(\tan(u)-u)\Bigg|_{0}^{\frac{\pi}{4}}-\int_{0}^{\frac{\pi}{4}}(\tan(u)-u)\mathrm{d}u\right)
\end{align*}

Now, 
\begin{align*} 
    \int_{0}^{\frac{\pi}{4}}(\tan(u)-u)\mathrm{d}u =  \int_{0}^{\frac{\pi}{4}}\tan(u)\mathrm{d}u-\int_{0}^{\frac{\pi}{4}}u\mathrm{d}u=\int_{0}^{\frac{\pi}{4}}\frac{\sin(u)}{\cos(u)}\mathrm{d}u-\frac{u^2}{2}\Bigg|_{0}^{\frac{\pi}{4}}
\end{align*}

Substituting $v=\cos(u)$, we get $\mathrm{d}v=-\sin(u)\mathrm{d}u$
\begin{align*}
    \int_{0}^{\frac{\pi}{4}}\frac{\sin(u)}{\cos(u)}\mathrm{d}u-\frac{u^2}{2}\Bigg|_{0}^{\frac{\pi}{4}}=-\int_{1}^{\frac{1}{\sqrt{2}}}\frac{\mathrm{d}v}{v}-\frac{u^2}{2}\Bigg|_{0}^{\frac{\pi}{4}}=-\ln(|\cos(u)|)\Bigg|_0^{\frac{\pi}{4}}-\frac{u^2}{2}\Bigg|_{0}^{\frac{\pi}{4}}
\end{align*}

Substituting this value back into the original integral, we get 
\begin{align*}  
    &-2\left(u(\tan(u)-u)\Bigg|_{0}^{\frac{\pi}{4}}-\int_{0}^{\frac{\pi}{4}}(\tan(u)-u)\mathrm{d}u\right) \\ &=-2\left(u(\tan(u)-u)\Bigg|_{0}^{\frac{\pi}{4}}-\left(-\ln(|\cos(u)|)\Bigg|_0^{\frac{\pi}{4}}-\frac{u^2}{2}\Bigg|_{0}^{\frac{\pi}{4}}\right)\right) \\ &=-2\left(\frac{\pi}{4}\left(1-\frac{\pi}{4}\right)-\left(-\ln\left(\frac{1}{\sqrt{2}}\right)-\frac{\pi^2}{32}\right)\right) \\
    &=-2\left(\frac{\pi}{4}-\frac{\pi^2}{16}-\frac{1}{2}\ln(2)+\frac{\pi^2}{32}\right) \\
    &=-2\left(\frac{\pi}{4}-\frac{1}{2}\ln(2)-\frac{\pi^2}{32}\right) \\
    &= -\frac{\pi}{2}+\ln(2)+\frac{\pi^2}{16}
\end{align*}

\begin{flalign*}
    (3)&&
\end{flalign*}


\begin{align*}
    -\int_{0}^{1}\frac{\ln(2)}{\alpha+1}\mathrm{d}\alpha=-\ln(2)\ln(|\alpha+1|)\Bigg|_{0}^{1}=-(\ln(2))^2.
\end{align*}

\vspace{20mm}
We add the values of the integrals together to get
\begin{align*}
    I &= \int_{0}^{1}\frac{\ln(\alpha+1)}{2(\alpha+1)}\mathrm{d}\alpha-\int_{0}^{1}\frac{\sqrt{\alpha}\arctan{\sqrt{\alpha}}}{1+\alpha}\mathrm{d}\alpha-\int_{0}^{1}\frac{\ln(2)}{\alpha+1}\mathrm{d}\alpha \\ 
    &= \frac{(\ln(2))^2}{4}-\frac{\pi}{2}+\ln(2)+\frac{\pi^2}{16}-(\ln(2))^2 \\
    &= \ln(2)-\frac{3\cdot(\ln(2))^2}{4}-\frac{\pi}{2}+\frac{\pi^{2}}{16}.
\end{align*}

Thus, it has been shown:

\begin{equation*}
    \sum_{n=1}^{\infty} \frac{(-1)^n}{n} \left(\frac{1}{2n-1} - \frac{1}{2n} + \frac{1}{2n+1}-\cdots\right)= \ln(2)-\frac{3\cdot(\ln(2))^2}{4}-\frac{\pi}{2}+\frac{\pi^{2}}{16}.
\end{equation*}
